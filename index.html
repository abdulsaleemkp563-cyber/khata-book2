Alarm check logic പരിഹരിച്ചത് function checkReminderAlerts() { if (!db.reminders) return; const now = new Date(); console.log('Checking reminders at:', now.toLocaleString()); db.reminders.forEach(rem => { if (rem.alertSent) return; // Parse date and time correctly const [year, month, day] = rem.dueDate.split('-').map(Number); const [hours, minutes] = rem.dueTime.split(':').map(Number); const reminderDateTime = new Date(year, month - 1, day, hours, minutes, 0); console.log('Reminder:', rem.title, 'Due:', reminderDateTime.toLocaleString()); // Check if it's time (allow 1 minute tolerance) const diffMs = reminderDateTime - now; const diffMinutes = Math.floor(diffMs / 60000); if (diffMinutes <= 0 && diffMinutes > -2) { // Time's up! console.log('ALARM TRIGGERED for:', rem.title); if ('Notification' in window && Notification.permission === 'granted') { new Notification('🔔 KHATA PRO Reminder', { body: `${rem.title}\nTime's up!`, icon: '🔔' }); } playAlarmSound(); rem.alertSent = true; saveDB(); } }); }
